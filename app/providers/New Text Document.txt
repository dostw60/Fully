from typing import Dict, List, Optional
from datetime import datetime, timedelta
import random

class MockProvider:
    """Mock provider that generates sample data when NEPSE API is down"""
    
    def __init__(self):
        self._mock_companies = self._generate_mock_companies()
        self._mock_stocks = self._generate_mock_stocks()
    
    def _generate_mock_companies(self):
        """Generate mock company list"""
        symbols = ['NABIL', 'HHL', 'NTC', 'SCB', 'NICA', 'KBL', 'NIBL', 'GBIME', 'NBL', 'SBI']
        companies = []
        for i, symbol in enumerate(symbols, start=1):
            companies.append({
                "id": i,
                "symbol": symbol,
                "name": f"{symbol} Limited",
                "sector": "Commercial Banks"
            })
        return companies
    
    def _generate_mock_stocks(self):
        """Generate mock stock data"""
        base_prices = {
            'NABIL': 525,
            'HHL': 338,
            'NTC': 650,
            'SCB': 480,
            'NICA': 412,
            'KBL': 206,
            'NIBL': 350,
            'GBIME': 280,
            'NBL': 150,
            'SBI': 420
        }
        
        stocks = []
        for symbol, base_price in base_prices.items():
            variation = random.uniform(-10, 10)
            current_price = max(base_price + variation, 1)
            change_percent = (current_price - base_price) / base_price * 100
            
            stocks.append({
                "securityId": str(base_prices.index(base_price) + 1),
                "securityName": f"{symbol} Limited",
                "symbol": symbol,
                "openPrice": base_price,
                "highPrice": current_price + abs(random.uniform(0, 5)),
                "lowPrice": current_price - abs(random.uniform(0, 5)),
                "totalTradeQuantity": random.randint(1000, 50000),
                "totalTradeValue": random.randint(100000, 10000000),
                "lastTradedPrice": current_price,
                "percentageChange": round(change_percent, 2),
                "lastUpdatedDateTime": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                "previousClose": base_price,
                "averageTradedPrice": current_price
            })
        
        return stocks
    
    def get_all_stocks(self) -> List[Dict]:
        """Get all mock stocks"""
        return self._mock_stocks
    
    def get_live_price(self, symbol: str) -> Dict:
        """Get mock live price for a company"""
        for stock in self._mock_stocks:
            if stock.get('symbol') == symbol:
                return {
                    "symbol": symbol,
                    "security_id": stock.get('securityId'),
                    "price": stock.get('lastTradedPrice', 0),
                    "change": stock.get('lastTradedPrice', 0) - stock.get('previousClose', 0),
                    "change_percent": stock.get('percentageChange', 0),
                    "volume": stock.get('totalTradeQuantity', 0),
                    "timestamp": datetime.now().isoformat(),
                    "open": stock.get('openPrice', 0),
                    "high": stock.get('highPrice', 0),
                    "low": stock.get('lowPrice', 0),
                    "prev_close": stock.get('previousClose', 0),
                    "turnover": stock.get('totalTradeValue', 0),
                    "company_name": stock.get('securityName', ''),
                    "sector": "Commercial Banks",
                    "isin": "NPE000A00000",
                    "face_value": 100,
                    "52_week_high": stock.get('highPrice', 0) * 1.2,
                    "52_week_low": stock.get('lowPrice', 0) * 0.8,
                    "last_updated": stock.get('lastUpdatedDateTime', '')
                }
        return {"error": f"Symbol {symbol} not found", "symbol": symbol}
    
    def get_top_gainers(self, limit: int = 10) -> List[Dict]:
        """Get mock top gainers"""
        sorted_stocks = sorted(
            self._mock_stocks,
            key=lambda x: x.get('percentageChange', 0),
            reverse=True
        )
        return sorted_stocks[:limit]
    
    def get_top_losers(self, limit: int = 10) -> List[Dict]:
        """Get mock top losers"""
        sorted_stocks = sorted(
            self._mock_stocks,
            key=lambda x: x.get('percentageChange', 0)
        )
        return sorted_stocks[:limit]
    
    def get_market_status(self) -> Dict:
        """Get mock market status"""
        return {
            "isOpen": True,
            "message": "Market is open",
            "timestamp": datetime.now().isoformat()
        }
    
    def search_stocks(self, query: str) -> List[Dict]:
        """Search for stocks by symbol"""
        results = []
        query_upper = query.upper()
        for company in self._mock_companies:
            if query_upper in company.get('symbol', '').upper():
                results.append(company)
        return results[:20]
